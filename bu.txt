  monitor:
    image: citusdata/pg_auto_failover:v2.0-pg15
    container_name: monitor
    environment:
      PGDATA: /var/lib/postgres/pgaf
      PG_AUTOCTL_DEBUG: 1
      TZ: 'Europe/Moscow'
    command: pg_autoctl create monitor --ssl-self-signed --auth trust --run
    ports:
      - "5500:5432"
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U postgres -d pg_auto_failover"]
